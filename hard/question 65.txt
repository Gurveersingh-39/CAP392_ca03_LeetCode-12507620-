class Solution {
    public boolean isNumber(String s) {

        boolean digit = false;
        boolean dot = false;
        boolean exponent = false;
        boolean number = false;

        for (int i = 0; i < s.length(); i++) {

            char ch = s.charAt(i);

            if (ch >= '0' && ch <= '9') {
                digit = true;
                number = true;
            }

            else if (ch == '+' || ch == '-') {

                if (i != 0 && s.charAt(i - 1) != 'e' && s.charAt(i - 1) != 'E') {
                    return false;
                }

                number = false;
            }

            else if (ch == '.') {

                if (dot || exponent) {
                    return false;
                }

                dot = true;
            }

            else if (ch == 'e' || ch == 'E') {

                if (exponent || !digit) {
                    return false;
                }

                exponent = true;
                digit = false;
            }

            else {
                return false;
            }
        }

        return number && digit;
    }
}
